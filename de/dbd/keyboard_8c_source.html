107static char translate_keycode_to_ascii(keycode_t keycode) {
108 char ascii;
109
110 if (keycode.release || keycode.keycode >= 128U) {
111 return 0;
112 }

114 ascii = keyboard_ascii_map[keycode.keycode];
115 if (ascii == 0) {

260void keyboard_handle_scancode(uint8_t scancode) {
261 keycode_t keycode;
262 char ascii;
263
264 push_scancode_buffer(&scancode_buffer, scancode);
265 scancode2keycode(&scancode_buffer, &keycode_buffer);
266
267 while (keycode_buffer.count > 0U) {
268 keycode = pop_keycode_buffer(&keycode_buffer);
269
270 if (keycode.keycode == KEYCODE_LEFT_SHIFT) {
271 left_shift_pressed = !keycode.release;
272 continue;
273 }
274
275 if (keycode.keycode == KEYCODE_RIGHT_SHIFT) {
276 right_shift_pressed = !keycode.release;
277 continue;
278 }
279

281 caps_lock_enabled = !caps_lock_enabled;
282 continue;
283 }
284
285 ascii = translate_keycode_to_ascii(keycode);
286 if (ascii == 0) {
287 continue;
288 }

290 push_ascii_buffer(ascii);
